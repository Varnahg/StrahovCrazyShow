import {Simulation,initPhysics} from '../../shared/simulation';import {stepRigidWorld} from '../../shared/physics-step';import {movementFromKeys} from '../../shared/controls';import {SPAWN} from '../../shared/world';import {Vector3} from '@babylonjs/core/Maths/math.vector';import {writeFileSync} from 'node:fs';
await initPhysics();const sim=new Simulation(),actor=sim.add('camera-probe'),camera=new Vector3(SPAWN.x,SPAWN.y+3,SPAWN.z+8);let arm=8,record;for(let i=0;i<315;i++){const previous={...actor.position};actor.step({seq:i+1,...movementFromKeys(i>=45?-1:0,0,Math.PI),yaw:Math.PI,jump:false,run:false});stepRigidWorld(sim.world);const position=new Vector3(previous.x,previous.y,previous.z),target=position.add(new Vector3(0,.6,0));let desired=target.add(new Vector3(-Math.sin(Math.PI)*8,2+.28*4,-Math.cos(Math.PI)*8));let delta=desired.subtract(target).normalize(),length=sim.cameraDistance(target,desired,actor);arm=length<arm?length:arm+(length-arm)*(1-Math.exp(-5/60));desired=target.add(delta.scale(arm));Vector3.LerpToRef(camera,desired,1-Math.exp(-12/60),camera);const safe=sim.cameraDistance(target,camera,actor);delta=camera.subtract(target);if(safe<delta.length()-.005)camera.copyFrom(target.add(delta.normalize().scale(safe)));record={tick:i,actor:{...actor.position},camera:camera.asArray(),renderedFeet:{x:position.x,y:position.y-.8,z:position.z},cameraArm:arm,distanceFromChest:Vector3.Distance(camera,target),userFirstPerson:false,legacyBodyVisible:true};}console.log(JSON.stringify(record,null,2));writeFileSync('.cache/camera-obstruction-independent/before.json',JSON.stringify(record,null,2));sim.dispose();
