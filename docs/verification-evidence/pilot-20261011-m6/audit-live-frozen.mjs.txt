async function hashFile(path) {
  const hash = createHash('sha256');
  let bytes = 0;
  for await (const chunk of createReadStream(path)) { hash.update(chunk); bytes += chunk.length; }
  return { sha256: hash.digest('hex'), bytes };
}

    try {
      const actual = await hashFile(path);
      computed[task.tree][task.file] = actual.sha256;
      counts[task.tree]++;
      bytesRead[task.tree] += actual.bytes;
      if (actual.sha256 !== task.expected) mismatches.push({ tree: task.tree, file: task.file, expected: task.expected, actual: actual.sha256 });

  }
}));
