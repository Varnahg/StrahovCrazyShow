import { createHash } from 'node:crypto';
import { createReadStream } from 'node:fs';
import { readFile, writeFile, mkdir, copyFile } from 'node:fs/promises';
import { resolve, join, sep } from 'node:path';
import assert from 'node:assert/strict';

const root = resolve(import.meta.dirname, '..');
const frozen = join(root, '.cache/performance-snapshot/pilot-20261011-m4');
const destination = join(root, 'docs/verification-evidence/pilot-20261011-m4');
const manifestBytes = await readFile(join(frozen, 'source-manifest.json'));
const manifest = JSON.parse(manifestBytes);
const expectedId = '38776c10a70e2148b5b2b525d366cecd35c768e0116857e211a02e32a1150b9c';
const digest = bytes => createHash('sha256').update(bytes).digest('hex');
assert.equal(manifest.snapshotId, expectedId);
assert.equal(digest(JSON.stringify(manifest.files)), expectedId);
const files = Object.entries(manifest.files);
const pipeline = JSON.parse(await readFile(join(frozen,'pipeline-results.json'),'utf8'));
assert.equal(pipeline.sourceSnapshotId, expectedId);
assert.equal(pipeline.files, files.length);
const startedAt = new Date().toISOString();
const start = performance.now();
const mismatches = [], errors = [], computed = {};
let checked = 0, bytesRead = 0, cursor = 0;
async function hashFile(path) {
  const hash = createHash('sha256');
  let bytes = 0;
  for await (const chunk of createReadStream(path)) { hash.update(chunk); bytes += chunk.length; }
  return { sha256: hash.digest('hex'), bytes };
}
await Promise.all(Array.from({ length: 8 }, async () => {
  while (cursor < files.length) {
    const [file, expected] = files[cursor++];
    const path = resolve(frozen, file);
    assert.ok(path.startsWith(frozen + sep));
    try {
      const actual = await hashFile(path);
      computed[file] = actual.sha256;
      checked++;
      bytesRead += actual.bytes;
      if (actual.sha256 !== expected) mismatches.push({ file, expected, actual: actual.sha256 });
    } catch (error) { errors.push({ file, error: String(error) }); }
  }
}));
const computedId = digest(JSON.stringify(Object.fromEntries(files.map(([file]) => [file, computed[file] ?? null]))));
const check = {
  startedAt, capturedAt: new Date().toISOString(), elapsedSeconds: (performance.now()-start)/1000,
  sourceSnapshotId: expectedId, manifestSha256: digest(manifestBytes), files: files.length,
  auditScope: 'frozen-only', frozenFilesChecked: checked, liveFilesChecked: 0,
  frozenBytesRead: bytesRead, computedFrozenSourceSnapshotId: computedId,
  status: mismatches.length === 0 && errors.length === 0 && checked === files.length && computedId === expectedId ? 'passed' : 'failed',
  method: 'Independent SHA-256 streaming read of every frozen manifest path, including shared historical IPR reference files. No cached hashes or reference deduplication. Aggregate ID reconstructed in original manifest key order.',
  currentLiveComparison: 'Not performed. Live workspace contains subsequent postprocess scene-component registration fix and regression test and is not claimed to match M4.',
  historicalFreezeVerification: { capturedAt: manifest.capturedAt, method: manifest.method, source: 'source-manifest.json',
    scope: 'The freeze operation reported matching live-before, live-after and copied frozen hashes at its original time. This historical statement is distinct from the present frozen-only audit.' },
  excludedFromThisAudit: ['Documentation', 'build outputs', 'dependencies', 'unlisted paths', 'current live workspace'],
  testFilesChecked: files.filter(([file]) => file.startsWith('tests/')).length,
  historicalReferenceFilesChecked: files.filter(([file]) => file.startsWith('public/models/photogrammetry/')).length,
  mismatches, errors,
};
await mkdir(destination, { recursive: true });
const publishedFiles = ['source-manifest.json','pipeline-results.json','pipeline-test.log','pipeline-build.log','pipeline-test-integration.log'];
const publishedSha256 = {};
for (const file of publishedFiles) {
  await copyFile(join(frozen, file), join(destination, file));
  const [source, copy] = await Promise.all([hashFile(join(frozen,file)),hashFile(join(destination,file))]);
  assert.equal(copy.sha256, source.sha256);
  publishedSha256[file] = copy.sha256;
}
await writeFile(join(destination,'source-check.json'), JSON.stringify(check,null,2)+'\n');
const testLog = await readFile(join(destination,'pipeline-test.log'),'utf8');
const testCount = kind => {
  const match = testLog.match(new RegExp('^.*?\\b'+kind+' (\\d+)\\s*$', 'm'));
  assert.ok(match, 'Missing test log count: '+kind);
  return Number(match[1]);
};
const tests = { total: testCount('tests'), passed: testCount('pass'), failed: testCount('fail'),
  cancelled: testCount('cancelled'), skipped: testCount('skipped'), todo: testCount('todo') };
assert.equal(tests.total, tests.passed);
const results = {
  status: check.status === 'passed' && pipeline.stages.every(stage=>stage.exitCode===0) && tests.failed===0 ? 'passed' : 'failed',
  scope: 'M4 frozen technical checkpoint: tests, production build including TypeScript, isolated integration, and subsequent independent frozen-only SHA audit. Does not describe subsequent scene-component registration fix; GPU validation pending.',
  sourceSnapshotId: expectedId, files: files.length, tests,

  sourceCheck: 'source-check.json', pipeline: 'pipeline-results.json', publishedSha256,

  gpu: { status: 'pending', note: 'No completed matched GPU/FPS result for M4 is attached here. Earlier L2/M evidence is not a GPU result for M4.' },
  buildNotes: [
    'Build completed with exit 0; raw log retains chunk-size, dynamic-import and plugin-timing warnings.',

for (const file of ['component-contract-repro.json','component-contract-repro.ts.txt','component-contract-repro.log','masked-constructor-order-probe.log'])
  results.publishedSha256[file] = (await hashFile(join(destination,file))).sha256;
