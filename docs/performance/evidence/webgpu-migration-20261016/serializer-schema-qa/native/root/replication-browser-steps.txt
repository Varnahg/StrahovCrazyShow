async function observeReplicaReady(){ const until=Date.now()+40000; while(Date.now()<until){if(await replicaTab.playwright.getByRole('button',{name:'Svítilna (F)',exact:true}).isEnabled())return {ready:true,observedUTC:Date.now()}; await new Promise(r=>setTimeout(r,1000));}return {ready:false,observedUTC:null};}

async function setupReplica(){ await replicaTab.playwright.getByRole('button',{name:'Nastavení',exact:true}).click(); await replicaTab.playwright.getByRole('combobox',{name:'Náhled osvětlení',exact:true}).selectOption({label:'Noční náhled · demo'}); await replicaTab.playwright.getByRole('button',{name:'Zavřít panel',exact:true}).click(); await replicaTab.pressKey(null,'s',{durationMs:1900}); await replicaTab.pressKey(null,'e'); await replicaTab.getAXState({emit:false}); }

async function timedReplicaBenchmark(run,readyAt){ const target=readyAt+30000; if(Date.now()>target+1500)throw Error('Pre-benchmark schedule missed; preserve attempt without result'); await new Promise(r=>setTimeout(r,Math.max(0,target-Date.now()))); if(!await replicaTab.playwright.getByText('E · nastoupit do auta',{exact:true}).isVisible())throw Error('Entry position not confirmed'); await replicaTab.playwright.getByRole('button',{name:'Nastavení',exact:true}).click(); const startAt=Date.now(); await replicaTab.playwright.getByRole('button',{name:'Změřit FPS po dobu 10 sekund',exact:true}).click(); await replicaTab.playwright.getByRole('button',{name:'Zavřít panel',exact:true}).click(); await replicaTab.pressKey(null,'e'); await replicaTab.pressKey(null,'w',{durationMs:8000}); await replicaTab.pressKey(null,'space',{durationMs:2500}); await replicaTab.playwright.getByRole('button',{name:'Nastavení',exact:true}).click(); await replicaTab.getAXState({emit:false}); await replicaTab.playwright.getByText('Podrobnosti měření',{exact:true}).click(); const text=await replicaTab.playwright.locator('#benchmark-result').textContent(),b=JSON.parse(text),dir='C:/Users/rezac/.codex/worktrees/29fa/StrahovCrazyShow/.cache/webgpu-serializer-schema-qa-20261019/runs-f3-replica/'+run+'/'; await cpuFs.writeFile(dir+'benchmark.json',text); await cpuFs.writeFile(dir+'timing.json',JSON.stringify({readyObservedUTC:new Date(readyAt).toISOString(),benchmarkClickUTC:new Date(startAt).toISOString(),readyObservedToClickMs:startAt-readyAt,targetMs:30000},null,2)); return {run,source:b.sourceSnapshotId,fps:b.averageFps,p95:b.p95FrameMs,p99:b.p99FrameMs,max:Math.max(...b.frameIntervalsMs),over100:b.frameIntervalsMs.filter(x=>x>=100).length,ambient:b.ambient,observedIdleMs:startAt-readyAt}; }

async function finishReplica(run){await replicaTab.playwright.getByRole('button',{name:'Zavřít panel',exact:true}).click(); await replicaTab.getAXState({emit:false}); await cpuFs.writeFile('C:/Users/rezac/.codex/worktrees/29fa/StrahovCrazyShow/.cache/webgpu-serializer-schema-qa-20261019/runs-f3-replica/'+run+'/first-car.jpg',await replicaTab.screenshot({fullPage:false})); await replicaTab.playwright.getByRole('button',{name:'Opustit demo',exact:true}).click(); await replicaTab.getAXState({emit:false});}
